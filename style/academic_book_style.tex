% ------------------------------------------------------------------------------
% 7.1 Page Clearing & Mainmatter Patching (openany compliance)
% ------------------------------------------------------------------------------
% LaTeX book.cls unconditionally issues \cleardoublepage in \mainmatter,
% forcing Chapter 1 to start on an odd page and emitting an accidental blank
% verso page when frontmatter ends on an odd page (e.g. ToC on page 3).
% Patch \mainmatter to use \clearpage so Chapter 1 begins immediately on Page 4.
\makeatletter
\patchcmd{\mainmatter}{\cleardoublepage}{\clearpage}{}{%
  \renewcommand\mainmatter{%
    \clearpage\@mainmattertrue\pagenumbering{arabic}%
  }%
}
\makeatother

% 10.1 Wide-Spaced English Writing Lines (Default: 11mm for comfortable adult writing)
% Uses pure TeX counter loop and breakable paragraph rule lines for exact \linewidth alignment and 0pt overfull.
\newcount\wl@i
\newcommand{\writinglines}[2][11mm]{%
  \ifnum#2>0\relax
    \par\addvspace{2mm}%
    \begingroup
    \setlength{\parindent}{0pt}%
    \setlength{\parskip}{0pt}%
    \wl@i=0\relax
    \loop\ifnum\wl@i<#2\relax
      \advance\wl@i by 1\relax
      \vspace{\dimexpr#1-\baselineskip\relax}%
      \noindent\makebox[\linewidth]{\color{RuleLineColor}\rule{\linewidth}{0.5pt}}\par
    \repeat
    \endgroup
    \par\addvspace{2mm}%
  \fi
}

% 10.2 Adaptive Inline Fill-in-the-Blank Underline
% Measures prompt length to prevent overfull \hbox warnings.
\newlength{\fb@len}
\newcommand{\fillblank}[2][3.5cm]{%
  \settowidth{\fb@len}{\small\itshape #2}%
  \ifdim\fb@len>#1\relax
    \underline{\hspace{0.5em}{\small\color{AcadGray}\itshape #2}\hspace{0.5em}}%
  \else
    \underline{\makebox[#1][c]{\small\color{AcadGray}\itshape #2}}%
  \fi
}

% 10.4 Structured Sentence Rewrite Component
\newcommand{\SentenceRewriteItem}[3][2]{%
  \par\addvspace{8pt}%
  \noindent\textbf{\color{AcadNavy}#2}\enspace\textbf{Original:}\quad #3\par\vspace{3pt}%
  \noindent\textbf{\color{AcadForest}Revision:}\par
  \writinglines[11mm]{#1}%
}

% 10.5 Paragraph Revision Component
\newcommand{\current@pr@lines}{6}
\newenvironment{ParagraphRevision}[3][6]{%
  \def\current@pr@lines{#1}%
  \par\addvspace{10pt}%
  \noindent\textbf{\color{AcadForest}Case #2:}\enspace \textbf{#3}\par\vspace{4pt}%
}{%
  \par\addvspace{4pt}%
  \noindent\textbf{\color{AcadNavy}Revised Academic Paragraph:}\par
  \writinglines[11mm]{\current@pr@lines}%
  \par\addvspace{8pt}%
}
